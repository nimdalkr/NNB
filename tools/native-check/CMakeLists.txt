cmake_minimum_required(VERSION 3.24)
project(NnbNativeCheck LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use the Win32 generator for the NNB DLL")
endif()
add_executable(nnb-native-smoke main.cpp)
target_compile_features(nnb-native-smoke PRIVATE cxx_std_17)
target_compile_definitions(nnb-native-smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(nnb-native-smoke PRIVATE ../../BWAPILIB/include)
add_executable(nnb-policy-check policy.cpp)
target_compile_features(nnb-policy-check PRIVATE cxx_std_17)
target_include_directories(nnb-policy-check PRIVATE ../../Steamhammer/Source)
add_executable(nnb-build-catalog catalog.cpp)
target_compile_features(nnb-build-catalog PRIVATE cxx_std_17)
target_include_directories(nnb-build-catalog PRIVATE ../../BWAPILIB/include ../../Steamhammer/Source)
target_link_libraries(nnb-build-catalog PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../Release/BWAPILIB.lib")
