cmake_minimum_required(VERSION 3.24)
project(nnb_terrain LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE)
set(DEPS "${ROOT}/build/deps")
set(BWTA_SOURCE "${ROOT}/build/terrain-source")
file(GLOB API_SOURCE "${ROOT}/BWAPILIB/Source/*.cpp")
add_library(bwapi_types STATIC ${API_SOURCE} "${ROOT}/BWAPILIB/UnitCommand.cpp")
target_include_directories(bwapi_types PUBLIC "${ROOT}/BWAPILIB/include" "${ROOT}/BWAPILIB/Source")
target_compile_definitions(bwapi_types PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
file(GLOB TERRAIN_SOURCE "${BWTA_SOURCE}/src/*.cpp")
list(FILTER TERRAIN_SOURCE EXCLUDE REGEX "/(main|stdafx)\\.cpp$")
add_executable(nnb-terrain main.cpp ${TERRAIN_SOURCE})
target_include_directories(nnb-terrain PRIVATE "${BWTA_SOURCE}/interface" "${BWTA_SOURCE}/src" "${DEPS}/CGAL-5.6.1/include" "${DEPS}/boost_1_83_0" "${DEPS}/auxiliary/gmp/include" "${ROOT}/Steamhammer/Source")
target_compile_definitions(nnb-terrain PRIVATE OFFLINE NOMINMAX _CRT_SECURE_NO_WARNINGS BOOST_ALL_NO_LIB CGAL_HEADER_ONLY CGAL_NO_AUTOLINK)
target_compile_options(nnb-terrain PRIVATE /bigobj /W0 /fp:strict)
target_precompile_headers(nnb-terrain PRIVATE "${BWTA_SOURCE}/src/stdafx.h")
target_link_libraries(nnb-terrain PRIVATE bwapi_types "${DEPS}/auxiliary/gmp/lib/libgmp-10.lib" "${DEPS}/auxiliary/gmp/lib/libmpfr-4.lib")
add_custom_command(TARGET nnb-terrain POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DEPS}/auxiliary/gmp/lib/libgmp-10.dll" "${DEPS}/auxiliary/gmp/lib/libmpfr-4.dll" $<TARGET_FILE_DIR:nnb-terrain>)
file(GLOB CACHE_SOURCE "${ROOT}/BWTA/src/*.cpp")
add_executable(nnb-terrain-check main.cpp ${CACHE_SOURCE})
target_include_directories(nnb-terrain-check PRIVATE "${ROOT}/BWTA/interface" "${ROOT}/BWTA/src")
target_compile_definitions(nnb-terrain-check PRIVATE OFFLINE NNB_CACHE_CHECK NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(nnb-terrain-check PRIVATE bwapi_types)
